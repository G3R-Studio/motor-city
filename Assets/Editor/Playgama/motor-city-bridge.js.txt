(function () {
  'use strict';
  const key = 'motorCityBridgePrefs.v1';
  const env = window.motorCityBridge = {
    initialized: false, language: 'en', audioEnabled: true, paused: false,
    prefs: Object.create(null), lastSaved: '{}', saving: false, retry: null,
    adRequest: null,
    flush: function () {
      if (!env.initialized || env.saving) return;
      const snapshot = JSON.stringify(env.prefs);
      if (snapshot === env.lastSaved) return;
      clearTimeout(env.retry);
      env.saving = true;
      Promise.resolve().then(() => bridge.storage.set([key], [snapshot]))
        .then(() => { env.lastSaved = snapshot; env.saving = false; env.flush(); })
        .catch(error => {
          console.warn('Motor City: Bridge storage write failed; retrying', error);
          env.saving = false;
          env.retry = setTimeout(env.flush, 3000);
        });
    }
  };
  function onAdState(type, state) {
    const request = env.adRequest;
    if (!request || request.type !== type) return;
    if (state === 'opened') request.opened = true;
    if (type === 'rewarded' && state === 'rewarded') request.earned = true;
    if (state === 'closed' || state === 'failed') request.finish();
  }
  function deadline(promise, stage) {
    let timer;
    return Promise.race([promise, new Promise((_, reject) => {
      timer = setTimeout(() => reject(new Error('Motor City: ' + stage + ' timed out')), 20000);
    })]).finally(() => clearTimeout(timer));
  }
  window.motorCityBridgeReady = Promise.resolve().then(() => {
    if (!window.bridge) throw new Error('Playgama Bridge script failed to load');
    return deadline(bridge.initialize(), 'Bridge initialization');
  }).then(async () => {
    env.language = bridge.platform.language || 'en';
    env.audioEnabled = bridge.platform.isAudioEnabled !== false;
    env.paused = bridge.platform.isPaused === true;
    bridge.platform.on(bridge.EVENT_NAME.PAUSE_STATE_CHANGED, value => {
      env.paused = !!value;
      if (env.notifyPause) env.notifyPause(env.paused);
    });
    bridge.platform.on(bridge.EVENT_NAME.AUDIO_STATE_CHANGED, value => {
      env.audioEnabled = !!value;
      if (env.notifyAudio) env.notifyAudio(env.audioEnabled);
    });
    bridge.advertisement.on(bridge.EVENT_NAME.REWARDED_STATE_CHANGED, state => onAdState('rewarded', state));
    bridge.advertisement.on(bridge.EVENT_NAME.INTERSTITIAL_STATE_CHANGED, state => onAdState('interstitial', state));

    // Passing false prevents automatic JSON conversion. Do not overwrite existing
    // progress if the initial read fails or the stored document cannot be decoded.
    const values = await deadline(bridge.storage.get([key], false), 'save read');
    const raw = values[0];
    if (raw != null && raw !== '') {
      const document = typeof raw === 'string' ? JSON.parse(raw) : raw;
      if (!document || typeof document !== 'object' || Array.isArray(document))
        throw new Error('Invalid save document');
      for (const k of Object.keys(document)) env.prefs[k] = String(document[k]);
    }
    env.lastSaved = JSON.stringify(env.prefs);
    env.initialized = true;
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) env.flush();
    });
    window.addEventListener('pagehide', env.flush);
  });
  // Attach a rejection observer immediately; the Unity startup gate displays it.
  window.motorCityBridgeReady.catch(error => console.error('Motor City: Bridge startup failed', error));
  window.motorCityBridgeStartupError = function (error) {
    console.error('Motor City: cannot start', error);
    const panel = document.createElement('div');
    panel.style.cssText = 'position:fixed;inset:0;display:grid;place-content:center;gap:1rem;background:#151b24;color:white;text-align:center;z-index:10000;font:18px sans-serif';
    const text = document.createElement('div');
    text.textContent = 'Unable to load the game or saved progress.';
    const button = document.createElement('button');
    button.textContent = 'Retry';
    button.onclick = () => location.reload();
    panel.append(text, button);
    document.body.appendChild(panel);
  };
})();
